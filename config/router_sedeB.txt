!==========================================================
! Router SedeB (1941) - Configuración completa
! Interfaces: G0/0 → LAN3, G0/1 → LAN4(Servidores), S0/0/0 → WAN
!==========================================================

enable
conf t

hostname SedeB

! --- Seguridad básica ---
enable secret Class123
service password-encryption
banner motd #ACCESO SOLO A PERSONAL AUTORIZADO#

! --- Dominio y usuario para SSHv2 ---
ip domain-name finalcomu.com
username admin secret Admin123
crypto key generate rsa modulus 2048
ip ssh version 2

! --- Consola ---
line console 0
 password cisco
 login
 exec-timeout 5 0
 exit

! --- VTY (solo SSH) ---
line vty 0 4
 login local
 transport input ssh
 exec-timeout 5 0
 exit

! ─── IPv4 interfaces ───────────────────────────────────
interface g0/0
 description LAN3 - 172.16.200.0/24
 ip address 172.16.200.1 255.255.255.0
 ip helper-address 172.16.200.1
 no shutdown

interface g0/1
 description LAN4 Servidores - 172.16.202.0/29
 ip address 172.16.202.1 255.255.255.248
 no shutdown

interface s0/0/0
 description Enlace WAN hacia SedeA
 ip address 172.16.202.10 255.255.255.252
 no shutdown

! ─── IPv6 interfaces ───────────────────────────────────
ipv6 unicast-routing

interface g0/0
 ipv6 address 2001:dbe:bef9:3::1/64
 ipv6 nd ra interval 30
 no shutdown

interface g0/1
 ipv6 address 2001:dbe:bef9:10::1/64
 no shutdown

interface s0/0/0
 ipv6 address 2001:dbe:bef9:ff01::2/64
 no shutdown

! ─── DHCP IPv4 para LAN3 ───────────────────────────────
ip dhcp excluded-address 172.16.200.1 172.16.200.10
ip dhcp pool LAN3
 network 172.16.200.0 255.255.255.0
 default-router 172.16.200.1
 dns-server 8.8.8.8
 domain-name finalcomu.com

! (LAN4/Servidores usan IPs estáticas, no DHCP)

! ─── Rutas estáticas IPv4 ──────────────────────────────
ip route 172.16.192.0 255.255.252.0 172.16.202.9
ip route 172.16.196.0 255.255.252.0 172.16.202.9

! ─── Rutas estáticas IPv6 ──────────────────────────────
ipv6 route 2001:dbe:bef9:1::/64 2001:dbe:bef9:ff01::1
ipv6 route 2001:dbe:bef9:2::/64 2001:dbe:bef9:ff01::1

end
write memory
